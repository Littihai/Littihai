const XSVG='<svg viewBox="0 0 50 50"><path d="M12 12L38 38M38 12L12 38"/></svg>';
function nodeHTML(n,useOld){
  const rep = n.mode==='replace';
  const ic = (rep&&useOld)? n.oldIc : n.ic;
  const th = (rep&&useOld)? n.oldTh : n.th;
  let meta;
  if(rep&&useOld)      meta=`<div class="meta">${L(n.oldEn||'')}</div>`;
  else if(rep)         meta=`<div class="meta ghost">${L(n.oldTh)}</div>`;
  else if(n.tag)       meta=`<span class="meta tag ${n.tag[1]}">${L(n.tag[0])}</span>`;
  else                 meta=`<div class="meta">${L(n.en||'')}</div>`;
  return `<div class="tile"><span class="ring"></span>${svg(ic)}</div>
    <span class="xmark">${XSVG}</span>
    <div class="lbl" title="${L(th)}">${L(th)}</div>${meta}`;
}
function build(cfg,rootId){
  const root=document.getElementById(rootId), sv=root.querySelector('.wires');
  cfg.root=root; cfg.sv=sv; cfg.map={}; cfg.emap={};
  cfg.nodes.forEach(n=>{
    const el=document.createElement('div');
    el.className='node'+(n.sm?' sm':'');
    el.style.left=n.x+'%'; el.style.top=n.y+'%';
    el.innerHTML=nodeHTML(n,true);
    root.appendChild(el); cfg.map[n.id]=el; n.el=el; n.r=n.sm?26:30;
  });
  cfg.edges.forEach(e=>{
    const g=document.createElementNS('http://www.w3.org/2000/svg','path');
    g.setAttribute('class','wire guide'); sv.appendChild(g); e.guide=g;
    const p=document.createElementNS('http://www.w3.org/2000/svg','path');
    const k=e.kind||'digital';
    p.setAttribute('class','wire'+(k==='back'?' back manual':' '+k));
    sv.appendChild(p); e.path=p; cfg.emap[e.id]=e;
  });
  layout(cfg);
}
function zig(guide,amp,wave){
  const L=guide.getTotalLength(); if(!L) return guide.getAttribute('d');
  const n=Math.max(4,Math.round(L/wave)); let d='';
  for(let i=0;i<=n;i++){
    const t=i/n, pt=guide.getPointAtLength(t*L);
    const a=guide.getPointAtLength(Math.max(0,t*L-1)), b=guide.getPointAtLength(Math.min(L,t*L+1));
    const dx=b.x-a.x, dy=b.y-a.y, m=Math.hypot(dx,dy)||1;
    const off=(i===0||i===n)?0:((i%2?1:-1)*amp);
    const x=pt.x+(-dy/m)*off, y=pt.y+(dx/m)*off;
    d+=(i?'L':'M')+x.toFixed(1)+','+y.toFixed(1);
  }
  return d;
}
function layout(cfg){
  const P=id=>{const e=cfg.map[id];return{x:e.offsetLeft,y:e.offsetTop};};
  const R=id=>cfg.nodes.find(n=>n.id===id).r;
  cfg.edges.forEach(e=>{
    const A=P(e.a),B=P(e.b);
    const dx=B.x-A.x,dy=B.y-A.y,d=Math.hypot(dx,dy)||1,ux=dx/d,uy=dy/d;
    const x1=A.x+ux*R(e.a),y1=A.y+uy*R(e.a),x2=B.x-ux*R(e.b),y2=B.y-uy*R(e.b);
    const mx=(x1+x2)/2,my=(y1+y2)/2,bow=e.bow||0;
    const dd=`M${x1},${y1} Q${mx+(-uy)*bow},${my+ux*bow} ${x2},${y2}`;
    e.guide.setAttribute('d',dd);
    e.path.setAttribute('d', (e.kind==='digital')? zig(e.guide,3,9) : dd);
  });
}

function anim(dur,cb){return new Promise(res=>{let t=0,last=performance.now();
  (function step(now){const dt=now-last;last=now;if(!paused)t+=dt*speed;
   const p=Math.min(1,t/dur);cb(p);p>=1?res():requestAnimationFrame(step);})(last);});}
const sleep=d=>anim(d,()=>{});

async function hop(cfg,e,dur){
  const path=e.path, track=e.guide, L=track.getTotalLength();
  path.classList.add('active');
  const g=particle(cfg);
  await anim(dur,p=>{
    const ease=cfg.side==='before'?p:(p<.5?2*p*p:1-Math.pow(-2*p+2,2)/2);
    const pt=track.getPointAtLength(ease*L);
    const rot=cfg.side==='before'?Math.sin(p*13)*6:0;
    g.setAttribute('transform',`translate(${pt.x},${pt.y}) rotate(${rot})`);
  });
  g.remove(); path.classList.remove('active'); path.classList.add('done');
}
async function swapStep(cfg,id,my){
  const n=cfg.nodes.find(v=>v.id===id), el=cfg.map[id];
  if(!n||n.mode!=='replace') return;
  el.classList.add('cut');
  await sleep(760); if(my!==token) return;
  el.classList.remove('cut');
  el.innerHTML=nodeHTML(n,false);
  el.classList.add('act','swap');
  setTimeout(()=>el.classList.remove('swap'),520);
}

}

async function runSeq(cfg,timerEl,msgEl,flagEl,my){
  for(const s of cfg.seq){
    if(my!==token) return;
    if(s.hop){ await Promise.all(s.hop.map(id=>hop(cfg,cfg.emap[id],s.dur||760))); continue; }
    if(s.t){ timerEl.textContent=L(s.t); timerEl.classList.toggle('small',!!s.ts); }
    if(s.m){ msgEl.textContent=L(s.m); msgEl.classList.toggle('warn',!!s.alarm); }
    activate(cfg,s.n,s);
    if(s.swap){ await swapStep(cfg,s.n[0],my); if(my!==token) return; }
    if(s.open) openPR();
    if(s.info) showInfo();
    if(s.ticket) showNo();
    if(s.chain) showChain();
    if(s.step0) stepTo(0);
    if(s.ap1) apOk(1,'09:40');
    if(s.ap2) apOk(2,'11:05');
    if(s.step3) stepTo(2);
    if(s.ap3) apOk(3,'14:20');
    if(s.close) closePO();
    if(s.dash) showDash();
    if(s.hot) hotDash();
    if(s.audit) kpiCard.classList.add('reveal');
    await sleep(s.dw||600);
  }
  if(my!==token) return;
  flagEl.classList.add('show');
}

function resetSide(cfg,timerEl,msgEl,flagEl){
  cfg.nodes.forEach(n=>{ n.el.className='node'+(n.sm?' sm':''); n.el.innerHTML=nodeHTML(n,true); });
  cfg.edges.forEach(e=>e.path.classList.remove('done','active'));
  cfg.sv.querySelectorAll('g').forEach(g=>g.remove());
  timerEl.textContent='—'; timerEl.classList.remove('small');
  msgEl.textContent=L('รอเริ่มกระบวนการ'); msgEl.classList.remove('warn');
  flagEl.classList.remove('show');
}
function resetUI(){
  tcat.classList.remove('on'); tNo.textContent='— — —'; tTitle.textContent=L('รอสร้างใบขอซื้อ');
  setSt(L('ร่าง'),'');
  fb1.classList.remove('in'); fb2.classList.remove('in');
  [1,2,3].forEach(i=>{ document.getElementById('ap'+i).classList.remove('in','ok');
    document.getElementById('apt'+i).textContent=L('รอ');
    document.getElementById('al'+i).classList.remove('in'); });
  cBtn.className='cbtn'; cBtn.textContent=L('รออนุมัติครบทุกลำดับ');
  stps.forEach(s=>s.classList.remove('done','now'));
  barOld.style.width='0%'; barNew.style.width='0%'; vOld.textContent='—'; vNew.textContent='—';
  ['h1','h2','h3','h4'].forEach(id=>document.getElementById(id).classList.remove('in','hot'));
  kpiCard.classList.remove('reveal');
}
async function play(){
  const my=++token;
  resetSide(BEFORE,tB,mB,sB); resetSide(AFTER,tA,mA,sA); resetUI();
  await sleep(500); if(my!==token) return;
  await Promise.all([runSeq(BEFORE,tB,mB,sB,my), runSeq(AFTER,tA,mA,sA,my)]);
  if(my!==token) return;
  kpiCard.classList.add('reveal');
  await sleep(5600); if(my!==token) return;
  play();
}
